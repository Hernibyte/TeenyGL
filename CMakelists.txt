cmake_minimum_required(VERSION 3.5...4.0)
project(TeenyGL VERSION 0.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIGURATION>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIGURATION>")

option(TEENYGL_BUILD_EXAMPLES "Build TinyGL example programs" ${PROJECT_IS_TOP_LEVEL})

# ========================
# CMake Dependencies
# ========================
include(FetchContent)

# ========================
# Project Dependencies
# ========================

# glfw
if (NOT TARGET glfw)
    FetchContent_Declare(
        glfw
        GIT_REPOSITORY https://github.com/glfw/glfw.git
        GIT_TAG 3.4
        GIT_SHALLOW TRUE
    )

    # 1. Configure GLFW options BEFORE declaring or making available
    set(GLFW_BUILD_EXAMPLES OFF CACHE INTERNAL "Build the GLFW example programs")
    set(GLFW_BUILD_TESTS OFF CACHE INTERNAL "Build the GLFW test programs")
    set(GLFW_BUILD_DOCS OFF CACHE INTERNAL "Build the GLFW documentation")
    set(GLFW_INSTALL OFF CACHE INTERNAL "Generate installation target")

    FetchContent_MakeAvailable(glfw)
endif()

# glm
if (NOT TARGET glm)
    FetchContent_Declare(
        glm
        GIT_REPOSITORY https://github.com/g-truc/glm.git
        GIT_TAG 1.0.1
        GIT_SHALLOW TRUE
    )
    FetchContent_MakeAvailable(glm)
endif()

# spdlog
if (NOT TARGET spdlog)
    FetchContent_Declare(
        spdlog
        GIT_REPOSITORY https://github.com/gabime/spdlog.git
        GIT_TAG v1.15.0
        GIT_SHALLOW TRUE
    )
    FetchContent_MakeAvailable(spdlog)
endif()

# stb_image
if (NOT TARGET stb_image)
    FetchContent_Declare(
        stb_image
        GIT_REPOSITORY https://github.com/nothings/stb.git
        GIT_TAG master
        GIT_SHALLOW TRUE
    )
    FetchContent_MakeAvailable(stb_image)
    
    add_library(stb_image INTERFACE)
    target_include_directories(stb_image INTERFACE ${stb_image_SOURCE_DIR})
endif()

# glad
add_subdirectory(third_party/glad)

# ========================
# Examples
# ========================
if (TEENYGL_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()

# ========================
# Librería principal: TinyGL
# ========================
add_library(TeenyGL STATIC)
add_library(TeenyGL::TeenyGL ALIAS TeenyGL)

target_compile_features(TeenyGL PUBLIC cxx_std_20)

target_compile_definitions(TeenyGL
PRIVATE
    SPDLOG_COMPILED_LIB
    TGL_ENABLE_ASSERTS
    GLM_ENABLE_EXPERIMENTAL
)

target_sources(TeenyGL
    PUBLIC
        FILE_SET HEADERS
        BASE_DIRS ./include
        FILES
            # Root
            include/TeenyGL.h
            
            # Core
            include/core/render_window.h
            include/core/core.h
            include/core/timestep.h
            include/core/window.h

            # Events
            include/events/event.h
            include/events/app_event.h
            include/events/key_event.h
            include/events/mouse_event.h

            # Input
            include/input_code/mouse.h
            include/input_code/keyboard.h

            # Platform
            include/platform/gl/gl_buffers.h
            include/platform/gl/gl_context.h
            include/platform/gl/gl_renderer_api.h
            include/platform/gl/gl_shader.h
            include/platform/gl/gl_texture_2d.h
            include/platform/gl/gl_vertex_array.h
            include/platform/assert.h
            include/platform/default_types.h
            include/platform/log.h

            # Renderer
            include/renderer/render_objects/render_object_presets.h
            include/renderer/buffers.h
            include/renderer/camera.h
            include/renderer/context.h
            include/renderer/render_object.h
            include/renderer/renderer.h
            include/renderer/renderer_api.h
            include/renderer/shader.h
            include/renderer/texture.h
            include/renderer/vertex_array.h

    PRIVATE
        # Core
        src/core/render_window.cpp
        src/core/window.cpp
        
        # Platform
        src/platform/gl/gl_buffers.cpp
        src/platform/gl/gl_context.cpp
        src/platform/gl/gl_renderer_api.cpp
        src/platform/gl/gl_shader.cpp
        src/platform/gl/gl_texture_2d.cpp
        src/platform/gl/gl_vertex_array.cpp
        src/platform/log.cpp

        # Renderer
        src/renderer/render_objects/render_object_presets.cpp
        src/renderer/buffers.cpp
        src/renderer/camera.cpp
        src/renderer/context.cpp
        src/renderer/render_object.cpp
        src/renderer/renderer.cpp
        src/renderer/renderer_api.cpp
        src/renderer/shader.cpp
        src/renderer/texture.cpp
        src/renderer/vertex_array.cpp
)

# -----------------------------------------------------------------------------
# Include directories
# -----------------------------------------------------------------------------
target_include_directories(TeenyGL
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
)

target_link_libraries(TeenyGL
    PUBLIC
        glm::glm
        spdlog::spdlog
    PRIVATE
        glfw
        glad
        stb_image
)
